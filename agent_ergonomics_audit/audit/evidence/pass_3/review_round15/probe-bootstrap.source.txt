(require 'ert)
(require 'ogent-agent)
(require 'ogent-tools)
(require 'ogent-tool-process)
(require 'ogent-ui-toolcalls)

(defun review15--native (value)
  (if (stringp value)
      (json-parse-string value :object-type 'plist :null-object :json-null
                         :false-object :json-false)
    value))

(defun review15--file (root name content)
  (let ((path (expand-file-name name root)))
    (make-directory (file-name-directory path) t)
    (with-temp-file path (insert content))
    path))

(ert-deftest review15-read-pages-reconstruct-positions ()
  (let* ((root (make-temp-file "review15-read-" t))
         (ogent-tool-registry (copy-tree ogent-tools-default-registry))
         (ogent-ledger-enabled nil))
    (unwind-protect
        (dolist (text '("" "\n" "λ\n\n🙂界abcd\nΩ" "ab\ncd\n"))
          (let* ((file (review15--file root "λ.txt" text))
                 (expected (unless (string-empty-p text) (split-string text "\n"))))
            (when (string-suffix-p "\n" text) (setq expected (butlast expected)))
            (dolist (budget '(1 2 3 7))
              (dolist (format '(plist json))
                (let* ((ogent-tools-max-output-chars budget)
                       (parts (make-vector (length expected) ""))
                       (page (ogent-agent-call "read" (list :file_path file :limit 2) format))
                       (steps 0) snapshot)
                  (while (equal (plist-get (review15--native page) :status) "ok")
                    (should (< (cl-incf steps) 100))
                    (let* ((result (review15--native page)) (data (plist-get result :data)))
                      (should (<= (length (plist-get data :content)) budget))
                      (if snapshot (should (equal snapshot (plist-get data :snapshot)))
                        (setq snapshot (plist-get data :snapshot)))
                      (seq-doseq (line (plist-get data :lines))
                        (let* ((index (1- (plist-get line :number)))
                               (previous (aref parts index)))
                          (should (= (plist-get line :column) (1+ (length previous))))
                          (aset parts index (concat previous (plist-get line :text)))))
                      (when (eq format 'json)
                        (should (equal page (plist-get (json-parse-string
                                                      (json-serialize (list :nested page))
                                                      :object-type 'plist) :nested)))))
                    (setq page (ogent-agent-next page format)))
                  (should (equal (plist-get (review15--native page) :status) "done"))
                  (should (equal (append parts nil) expected))))))
      (delete-directory root t))))

(ert-deftest review15-search-pagination-scope-and-budget ()
  (let* ((root (make-temp-file "review15-search-" t))
         (ogent-tool-registry (copy-tree ogent-tools-default-registry))
         (ogent-ledger-enabled nil)
         (find-executable (symbol-function 'executable-find)))
    (unwind-protect
        (progn
          (dolist (name '("a.txt" ".hidden.txt" "目录/λ.txt" "colon:\n😀.txt"))
            (review15--file root name "before\nneedle λ\nafter\n"))
          (review15--file root "binary.txt" "needle\n\0tail\n")
          (dolist (engine '(gnu ripgrep))
            (cl-letf (((symbol-function 'executable-find)
                       (lambda (name &optional remote)
                         (if (equal name "rg")
                             (when (eq engine 'ripgrep) (getenv "OGENT_PROCESS_TEST_RG"))
                           (funcall find-executable name remote)))))
              (let* ((ogent-tools-max-output-chars 1000)
                     (whole (ogent-agent-call "search" (list :pattern "needle" :path root :context_lines 1)))
                     (expected (mapcar (lambda (x) (list (plist-get x :path) (plist-get x :line)))
                                       (append (plist-get (plist-get whole :data) :matches) nil))))
                (should (= (length expected) 4))
                (dolist (format '(plist json))
                  (let* ((ogent-tools-max-output-chars 3)
                         (page (ogent-agent-call "search" (list :pattern "needle" :path root :context_lines 1 :limit 1) format))
                         collected (steps 0))
                    (while (equal (plist-get (review15--native page) :status) "ok")
                      (should (< (cl-incf steps) 10))
                      (let* ((result (review15--native page)) (data (plist-get result :data)))
                        (should (= (plist-get data :total_matches) 4))
                        (should (equal (plist-get data :snapshot) (plist-get (plist-get whole :data) :snapshot)))
                        (seq-doseq (match (plist-get data :matches))
                          (push (list (plist-get match :path) (plist-get match :line)) collected))
                        (when (> (length (plist-get result :next)) 0)
                          (should (file-name-absolute-p
                                   (plist-get (plist-get (aref (plist-get result :next) 0) :args) :path)))))
                      (let ((ogent-tools-project-root "/unrelated/"))
                        (setq page (ogent-agent-next page format))))
                    (should (equal (plist-get (review15--native page) :status) "done"))
                    (should (equal (nreverse collected) expected))))))))
      (delete-directory root t))))

(ert-deftest review15-gptel-cached-tool-preserves-format-and-staleness ()
  (let* ((root (make-temp-file "review15-model-" t))
         (ogent-tool-registry (copy-tree ogent-tools-default-registry))
         (ogent-tools-result-format 'json)
         (ogent--tools-registered nil) (ogent--tool-specs-registered nil)
         (ogent--tool-formats-registered nil) (gptel--known-tools nil)
         (ogent-ledger-enabled nil))
    (unwind-protect
        (let* ((file (review15--file root "😀.txt" "λ one\nΩ two\n"))
               (tool (ogent-tool-get "read"))
               (result (funcall (gptel-tool-function tool) file 1 1))
               (envelope (review15--native result)))
          (should (gptel-tool-p tool))
          (should (equal (plist-get envelope :status) "ok"))
          (should (= (length (plist-get envelope :next)) 1))
          (let ((nested (json-parse-string (json-serialize (list :tool_result result)) :object-type 'plist)))
            (should (equal (plist-get nested :tool_result) result)))
          (setq ogent-tools-result-format 'text)
          (should-not (eq tool (ogent-tool-get "read")))
          (should (equal (plist-get (review15--native (funcall (gptel-tool-function tool) file 1 1)) :status) "ok"))
          (setf (plist-get (ogent-tool-spec-get 'read-file) :description) "Changed live metadata")
          (should (equal (plist-get (plist-get (review15--native
                                              (funcall (gptel-tool-function tool) file 1 1)) :error) :code)
                         "unavailable")))
      (delete-directory root t))))

(ert-deftest review15-async-json-validation-and-terminal-error-once ()
  (let* ((ogent-tool-registry (copy-tree ogent-tools-default-registry))
         (ogent-tool-require-approval t) (ogent-tool--denied-tools nil)
         (ogent-tool-allow-list nil) (ogent-ledger-enabled nil))
    (dolist (call '(("read" (:file_path 4) "invalid_arguments")
                    ("search" (:pattern "needle" :unknown t) "invalid_arguments")
                    ("shell" (:command "printf unsafe") "approval_required")
                    ("unknown" nil "unknown_tool")))
      (let ((count 0) result)
        (should-not (ogent-agent-call-async
                     (car call) (cadr call)
                     (lambda (data) (cl-incf count) (setq result (review15--native data))) 'json))
        (should (= count 1))
        (should (equal (plist-get (plist-get result :error) :code) (nth 2 call)))
        (should (equal (plist-get result :next) [])))))

(ert-run-tests-batch-and-exit "^review15-")
